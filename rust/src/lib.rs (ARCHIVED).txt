pub mod physics;

use pyo3::prelude::*;
use pyo3::types::PyDict;
use crate::physics::{GameState, Inputs, step};

#[pyclass]
struct RustEnv {
    state: GameState,
}

#[pymethods]
impl RustEnv {
    #[new]
    fn new(seed: u64) -> Self {
        Self { state: GameState::reset(seed, 0, "") }
    }

    #[pyo3(signature = (seed=None, stage=0, map_json=""))]
    fn reset(&mut self, seed: Option<u64>, stage: u8, map_json: &str) -> PyResult<PyObject> {
        self.state = GameState::reset(seed.unwrap_or(0), stage, map_json);
        Python::with_gil(|py| Ok(self.state.to_py(py)))
    }

    fn load_map(&mut self, json_str: &str) {
        self.state.load_map(json_str);
    }

    fn step(&mut self, action1: Vec<i8>, action2: Vec<i8>) -> PyResult<(PyObject, f64, bool)> {
        let inputs1 = Inputs { move_dir: action1[0] - 1, jump_dir: action1[1] - 1, heavy: action1[2] == 1, grapple: action1[3] == 1 };
        let inputs2 = Inputs { move_dir: action2[0] - 1, jump_dir: action2[1] - 1, heavy: action2[2] == 1, grapple: action2[3] == 1 };
        step(&mut self.state, &inputs1, &inputs2);
        let reward = self.state.reward();
        let done = self.state.is_done();
        Python::with_gil(|py| Ok((self.state.to_py(py), reward, done)))
    }

    // TEST HOOKS
    fn set_p1_pos(&mut self, x: f64, y: f64) {
        self.state.p1.x = x;
        self.state.p1.y = y;
    }
    fn set_ball_pos(&mut self, x: f64, y: f64) {
        self.state.ball.x = x;
        self.state.ball.y = y;
    }
}

impl GameState {
    fn to_py(&self, py: Python) -> PyObject {
        let d = PyDict::new_bound(py); 
        d.set_item("t", self.t).unwrap();
        d.set_item("current_reward", self.current_reward).unwrap();
        d.set_item("reward_shaping", self.reward_shaping).unwrap();
        d.set_item("reward_terminal", self.reward_terminal).unwrap();
        d.set_item("reward_attach", self.reward_attach).unwrap();
        d.set_item("phi", self.phi).unwrap();
        d.set_item("spawn_success", self.spawn_success).unwrap();
        d.set_item("map_radius", self.map_radius).unwrap();
        d.set_item("map_center_x", self.map_center_x).unwrap();
        d.set_item("map_center_y", self.map_center_y).unwrap();
        d.set_item("goal_x", self.goal_x).unwrap();
        d.set_item("goal_y", self.goal_y).unwrap();
        d.set_item("terminated", self.terminated).unwrap();
        d.set_item("truncated", self.truncated).unwrap();
        d.set_item("stage", self.stage).unwrap();
        
        let (_, _, is_ball) = crate::physics::get_closest_wall_point(&self.p1, &self.ball, &self.walls);
        let dx = self.ball.x - self.p1.x;
        let dy = self.ball.y - self.p1.y;
        let d_g = (dx*dx + dy*dy).sqrt();
        let in_range = if d_g <= (201.0 / 6.81) { 1.0 } else { 0.0 };
        
        d.set_item("grapple_target_dx", dx).unwrap();
        d.set_item("grapple_target_dy", dy).unwrap();
        d.set_item("grapple_target_dist", d_g).unwrap();
        d.set_item("grapple_is_ball", if is_ball { 1.0 } else { 0.0 }).unwrap();
        d.set_item("grapple_in_range", in_range).unwrap();

        let p1_dict = PyDict::new_bound(py);
        p1_dict.set_item("x", self.p1.x).unwrap(); p1_dict.set_item("y", self.p1.y).unwrap();
        p1_dict.set_item("vx", self.p1.vx).unwrap(); p1_dict.set_item("vy", self.p1.vy).unwrap();
        p1_dict.set_item("angle", self.p1.angle).unwrap(); p1_dict.set_item("angular_velocity", self.p1.angular_velocity).unwrap();
        p1_dict.set_item("force_x", self.p1.force_x).unwrap(); p1_dict.set_item("force_y", self.p1.force_y).unwrap();
        p1_dict.set_item("torque", self.p1.torque).unwrap(); p1_dict.set_item("mass", self.p1.mass).unwrap();
        p1_dict.set_item("inv_mass", self.p1.inv_mass).unwrap(); p1_dict.set_item("inertia", self.p1.inertia).unwrap();
        p1_dict.set_item("inv_inertia", self.p1.inv_inertia).unwrap(); p1_dict.set_item("linear_damping", self.p1.linear_damping).unwrap();
        p1_dict.set_item("angular_damping", self.p1.angular_damping).unwrap(); p1_dict.set_item("radius", self.p1.radius).unwrap();
        p1_dict.set_item("alive", self.p1.alive).unwrap(); p1_dict.set_item("grapple_attached", self.p1.grapple_attached).unwrap();
        p1_dict.set_item("anchor_x", self.p1.anchor_x).unwrap(); p1_dict.set_item("anchor_y", self.p1.anchor_y).unwrap();
        p1_dict.set_item("rope_length", self.p1.rope_length).unwrap();

        let p2_dict = PyDict::new_bound(py);
        p2_dict.set_item("x", self.p2.x).unwrap(); p2_dict.set_item("y", self.p2.y).unwrap();
        p2_dict.set_item("vx", self.p2.vx).unwrap(); p2_dict.set_item("vy", self.p2.vy).unwrap();
        p2_dict.set_item("angle", self.p2.angle).unwrap(); p2_dict.set_item("angular_velocity", self.p2.angular_velocity).unwrap();
        p2_dict.set_item("force_x", self.p2.force_x).unwrap(); p2_dict.set_item("force_y", self.p2.force_y).unwrap();
        p2_dict.set_item("torque", self.p2.torque).unwrap(); p2_dict.set_item("mass", self.p2.mass).unwrap();
        p2_dict.set_item("inv_mass", self.p2.inv_mass).unwrap(); p2_dict.set_item("inertia", self.p2.inertia).unwrap();
        p2_dict.set_item("inv_inertia", self.p2.inv_inertia).unwrap(); p2_dict.set_item("linear_damping", self.p2.linear_damping).unwrap();
        p2_dict.set_item("angular_damping", self.p2.angular_damping).unwrap(); p2_dict.set_item("radius", self.p2.radius).unwrap();
        p2_dict.set_item("alive", self.p2.alive).unwrap(); p2_dict.set_item("grapple_attached", self.p2.grapple_attached).unwrap();
        p2_dict.set_item("anchor_x", self.p2.anchor_x).unwrap(); p2_dict.set_item("anchor_y", self.p2.anchor_y).unwrap();
        p2_dict.set_item("rope_length", self.p2.rope_length).unwrap();

        let ball_dict = PyDict::new_bound(py);
        ball_dict.set_item("x", self.ball.x).unwrap(); ball_dict.set_item("y", self.ball.y).unwrap();
        ball_dict.set_item("vx", self.ball.vx).unwrap(); ball_dict.set_item("vy", self.ball.vy).unwrap();
        ball_dict.set_item("radius", self.ball.radius).unwrap(); ball_dict.set_item("exists", self.ball.exists).unwrap();

        d.set_item("p1", p1_dict).unwrap();
        d.set_item("p2", p2_dict).unwrap();
        d.set_item("ball", ball_dict).unwrap();
        d.into()
    }
}

#[pymodule]
fn wdb_engine(m: &Bound<'_, PyModule>) -> PyResult<()> {
    m.add_class::<RustEnv>()?;
    Ok(())
}
